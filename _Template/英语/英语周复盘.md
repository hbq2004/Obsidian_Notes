```dataviewjs
const cur = dv.current();
const start = dv.date(cur.start);
const end = dv.date(cur.end);
const asNumber = value => {
  const number = Number(value);
  return Number.isFinite(number) ? number : 0;
};
const asDate = value => {
  if (!value) return null;
  try {
    const date = dv.date(value);
    return date && date.isValid !== false ? date : null;
  } catch (_) {
    return null;
  }
};
const inRange = p => {
  const date = asDate(p.date);
  return date && date >= start && date <= end;
};
const sessions = dv.pages('"03 - 英语/02 - 真题复盘/复习记录"').where(p => p.type === "english-session" && inRange(p)).array();
const reads = dv.pages('"03 - 英语/02 - 真题复盘/阅读"').where(p => p.type === "english-reading" && inRange(p)).array();
const essays = dv.pages('"03 - 英语/02 - 真题复盘/作文"').where(p => p.type === "english-writing" && inRange(p)).array();
const sum = (xs, key) => xs.reduce((n, x) => n + asNumber(x[key]), 0);
const c = sum(reads, "correct"), t = sum(reads, "total");
const days = new Set(sessions.map(p => String(p.date))).size;
dv.table(["学习天数", "有效分钟", "阅读篇数", "首做正确率", "作文复盘"], [[
  days, sum(sessions, "minutes"), reads.length, t ? `${Math.round(c / t * 100)}%` : "-", essays.length
]]);
```
