// 0. ★ 终极视觉优化：动态注入 CSS，彻底干掉多图状态下的列表黑点
dv.container.classList.add("clean-table");
let style = document.createElement("style");
style.innerHTML = `
    .clean-table ul { list-style-type: none !important; padding-left: 0 !important; margin: 0 !important; }
    .clean-table li { margin-bottom: 12px !important; } /* 多张图片之间留一点呼吸感 */
`;
dv.container.appendChild(style);

// 1. 获取当前笔记及标签
const currentFile = dv.current();
const currentTags = currentFile.file.tags;

if (!currentTags || currentTags.length === 0) {
    dv.paragraph("💡 **提示**：当前笔记没有标签。请添加 tags 以便自动推荐同类题。");
    return;
}

// 2. 构造基础查询：包含当前所有标签，且排除当前笔记自身
const query = currentTags.join(" and ");
let pages = dv.pages(query).where(p => p.file.path !== currentFile.file.path);

// 3. 弱点狙击模式！
let isDowngraded = currentFile.Notice !== undefined && String(currentFile.Notice).includes("降级警告");
if (isDowngraded) {
    pages = pages.where(p => p.Notice !== undefined && String(p.Notice).includes("降级警告"));
}

const result = [];
const imgRegex = /!\[\[(.*?)\]\]/g;

// 4. 遍历读取目标文章正文
for (let page of pages) {
    const content = await dv.io.load(page.file.path);
    if (content) {
        const matches = content.match(imgRegex);
        if (matches) {
            let imgEmbeds = [];
            for (let match of matches) {
                if (match.includes("题目")) {
                    let innerText = match.replace(/^!\[\[/, "").replace(/\]\]$/, "");
                    let fileName = innerText.split("|")[0].trim();
                    imgEmbeds.push(dv.fileLink(fileName, true));
                }
            }
            
            if (imgEmbeds.length > 0) {
                // 即使是多张图保留了数组格式，上面的 CSS 也会把黑点藏起来
                let displayData = imgEmbeds.length === 1 ? imgEmbeds[0] : imgEmbeds;
                
                let selectEl = document.createElement("select");
                selectEl.style.padding = "4px 8px";
                selectEl.style.borderRadius = "5px";
                selectEl.style.cursor = "pointer";
                selectEl.style.backgroundColor = "var(--interactive-normal)";
                selectEl.style.color = "var(--text-normal)";
                selectEl.style.border = "1px solid var(--background-modifier-border)";
                
                let currentLevel = page.level !== undefined && page.level !== null ? parseInt(page.level) : -1;
                
                const levelOptions = [
                    { val: -1, text: "- 暂未评级 -" },
                    { val: 0, text: "0级 (无法看懂)" },
                    { val: 1, text: "1级 (首次看懂)" },
                    { val: 2, text: "2级 (首次模仿)" },
                    { val: 3, text: "3级 (查资料做对)" },
                    { val: 4, text: "4级 (独立做对)" },
                    { val: 5, text: "5级 (熟练秒杀)" }
                ];
                
                levelOptions.forEach(opt => {
                    let optionEl = document.createElement("option");
                    optionEl.value = opt.val;
                    optionEl.text = opt.text;
                    if (opt.val === currentLevel) {
                        optionEl.selected = true;
                    }
                    selectEl.appendChild(optionEl);
                });
                
                selectEl.onchange = async (e) => {
                    let newLevel = parseInt(e.target.value);
                    if (newLevel === currentLevel) return;
                    
                    let tfile = app.vault.getAbstractFileByPath(page.file.path);
                    if (tfile) {
                        await app.fileManager.processFrontMatter(tfile, fm => {
                            fm.level = newLevel;
                            if (currentLevel !== -1 && newLevel < currentLevel) {
                                fm.Notice = "❗降级警告！";
                            } else {
                                if (fm.Notice) delete fm.Notice;
                            }
                        });
                        
                        if (currentLevel !== -1 && newLevel < currentLevel) {
                            new Notice(`⚠️ 发生降级：已降至 ${newLevel} 级，并在原笔记写入 Notice 警告！`, 4000);
                        } else {
                            new Notice(`✅ 评级提升至 ${newLevel} 级！`);
                        }
                    }
                };

                result.push([page.file.link, displayData, selectEl]);
            }
        }
    }
}

// 5. 动态渲染表头与表格
if (result.length > 0) {
    let headerText = isDowngraded ? "🚨 弱点突破：同类降级错题推荐" : "🔗 同类题推荐拓展";
    dv.header(3, headerText);
    dv.table(["相关文章", "题目图片预览", "状态反馈 (动态更新)"], result);
} else {
    if (pages.length === 0) {
        if (isDowngraded) {
            dv.paragraph(`🎉 **太棒了！** 虽然当前题目降级了，但在 \`${query}\` 考点下，你没有其他带有警告的降级错题了！`);
        } else {
            dv.paragraph(`✅ 题库中暂时没有与当前标签完全匹配的其他同类题。`);
        }
    } else {
        dv.paragraph(`✅ 找到了匹配的关联文章，但它们没有包含『题目』的图片。`);
    }
}
