1. $O(n)$ 扫一遍，然后删除即可
2. 注意可能有多个重复的节点，需要开哈希表记录节点值



时空复杂度：$O(n, n)$ 


```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */

const int N = 20000 + 5;

class Solution {
public:
    ListNode* removeDuplicateNodes(ListNode* head) {
        int S[N] = {0};
        ListNode *now = head, *pre = head;
        while (now != NULL) {
            int v = now->val;
            if (S[v] == 1) {
                pre->next = now->next;
            } else {
                S[v] = 1;
                pre = now;
            }
            now = now->next;
        }
        return head;
    }
};
```
