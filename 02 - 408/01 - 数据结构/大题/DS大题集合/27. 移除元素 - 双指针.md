[27. 移除元素](https://leetcode.cn/problems/remove-element/)

```cpp
class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        // 前后指针，把所有 val 移动到最后
        int n = nums.size();
        int i = 0, j = n - 1;
        while (i <= j) { // 边界问题，这里不能写 i < j，spj nums = [1], val = 1
            while (j >= 0 && nums[j] == val) j--;
            while (i < n && nums[i] != val) i++;
            if (i <= j) {
                swap(nums[i], nums[j]);
                i++;
                j--;
            }
        }
        return n - (n - 1 - (j + 1) + 1);
    }
};
```

```cpp
class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        int stack_size = 0;
        for (int x : nums) {
            if (x != val) {
                nums[stack_size++] = x; // 把 x 入栈
            }
        }
        return stack_size;
    }
};
```
